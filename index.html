⚙️ Връзка с Google Drive (за учителя)

Получавай аудио записите и резултатите на Мартин директно в твоя Drive

⚠️ Все още няма въведен адрес към Google Drive. Постави Web App URL от Google Apps Script по-долу, за да заработи бутонът „Изпрати на мис“.

Когато изпратиш готовия линк с параметър ?endpoint=... на Мартин, приложението ще се свърже автоматично с твоя Drive без той да настройва нищо.

📜 Готов код за Google Apps Script (настройка за 2 мин.)

  1. Отвори script.google.com → New Project и постави кода по-долу.
  2. Натисни Deploy → New deployment → Web app.
  3. Избери Execute as: Me и Who has access: Anyone → Deploy и копирай URL-а горе.
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var folderName = "Мисия БЕЛ - Мартин";
    var folders = DriveApp.getFoldersByName(folderName);
    var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);

    var stamp = (data.date || new Date().toISOString()).replace(/[:T]/g, "-").slice(0, 16);
    var readingUrl = "";
    var oralUrl = "";

    if (data.readingAudioBase64) {
      var rMime = data.readingAudioMime || "audio/webm";
      var rExt = rMime.indexOf("mp4") !== -1 ? "m4a" : "webm";
      var rBlob = Utilities.newBlob(Utilities.base64Decode(data.readingAudioBase64), rMime, "Chetene_" + stamp + "." + rExt);
      readingUrl = folder.createFile(rBlob).getUrl();
    }

    if (data.oralAudioBase64) {
      var oMime = data.oralAudioMime || "audio/webm";
      var oExt = oMime.indexOf("mp4") !== -1 ? "m4a" : "webm";
      var oBlob = Utilities.newBlob(Utilities.base64Decode(data.oralAudioBase64), oMime, "Ustni_" + stamp + "." + oExt);
      oralUrl = folder.createFile(oBlob).getUrl();
    }

    var sheetName = "Отчети - Мартин БЕЛ";
    var files = folder.getFilesByName(sheetName);
    var ss = files.hasNext() ? SpreadsheetApp.open(files.next()) : SpreadsheetApp.create(sheetName);
    if (!files.hasNext()) {
      DriveApp.getFileById(ss.getId()).moveTo(folder);
      ss.getActiveSheet().appendRow([
        "Дата", "Ученик", "Ниво", "XP Днес", "Общо XP",
        "Изпълнени мисии", "Диктовка (%)", "Диктовка текст",
        "Правила", "Конструктор", "Запис Четене", "Запис Устни"
      ]);
    }

    var xpObj = data.xp || {};
    ss.getActiveSheet().appendRow([
      data.date || "",
      data.student || "Мартин",
      data.level || "",
      xpObj.today || 0,
      xpObj.total || 0,
      (data.completedMissions || []).join(", "),
      data.dictationScore !== null && data.dictationScore !== undefined ? data.dictationScore + "%" : "-",
      data.dictationText || "",
      data.rulesScore || "-",
      data.constructorScore || "-",
      readingUrl,
      oralUrl
    ]);

    return ContentService.createTextOutput(JSON.stringify({ status: "ok" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
